bash

cat << 'EOF' > osint_workflow.sh
#!/bin/bash

TARGET_NAME="YajairaArosemena"
PHONE="+50769674636"
OUTDIR="output/osint_${TARGET_NAME}_$(date +%Y%m%d_%H%M)"
mkdir -p "$OUTDIR"

echo "[*] Starting OSINT Workflow for $TARGET_NAME"

python3 sherlock/sherlock.py "$TARGET_NAME" > "$OUTDIR/sherlock_${TARGET_NAME}.txt"
maigret "$TARGET_NAME" -j -o "$OUTDIR/maigret_${TARGET_NAME}.json"
twint -u "$TARGET_NAME" -o "$OUTDIR/twint_${TARGET_NAME}.json" --json
curl "http://apilayer.net/api/validate?access_key=YOUR_KEY&number=${PHONE}" > "$OUTDIR/numverify_${TARGET_NAME}.json"

adb logcat -v time > "$OUTDIR/device_log_${TARGET_NAME}.txt" &
sleep 30
adb shell input keyevent 66
kill $!

echo "[*] Manual tasks: Run WhatsApp/LinkedIn scrapers & Maltego import"
EOF

chmod +x osint_workflow.sh
